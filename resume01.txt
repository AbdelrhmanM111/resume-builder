  async function generateResume() {
    var data = collectData();
    if (!data.personal.firstName) { showToast('Please complete Step 1 first', 'error'); return; }

    if (currentProvider === 'local') {
      showLoading(true);
      setTimeout(function() {
        try {
          var resumeData = buildLocalResume(data);
          renderResume(data.personal, resumeData);
          showToast('Resume created instantly! ✨', 'success');
          gaEvent('resume_generated', { method: 'local_template', tone: data.tone }); // ✅ GA Event
        } catch(e) {
          showToast('Error: ' + e.message, 'error');
        } finally {
          showLoading(false);
        }
      }, 800);
      return;
    }

    var apiKey = document.getElementById('api-key').value.trim();
    if (!apiKey) {
      showToast('Please enter your API key, or switch to Local Template (no key needed)', 'error');
      return;
    }

    showLoading(true);
    try {
      var prompt = buildPrompt(data);
      var content;

      if (currentProvider === 'gemini') {
        var geminiRes = await fetch(
          'https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=' + apiKey,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              contents: [{ parts: [{ text: 'You are a professional resume writer. Always respond with valid JSON only, no markdown.\n\n' + prompt }] }],
              generationConfig: { temperature: 0.7, maxOutputTokens: 2000 }
            })
          }
        );
        if (!geminiRes.ok) {
          var geminiErr = await geminiRes.json();
          throw new Error((geminiErr.error && geminiErr.error.message) || 'Gemini API error');
        }
        var geminiData = await geminiRes.json();
        content = geminiData.candidates[0].content.parts[0].text.trim();

      } else {
        // OpenAI
        if (!apiKey.startsWith('sk-')) { showToast('Invalid OpenAI key format', 'error'); showLoading(false); return; }
        var openaiRes = await fetch('https://api.openai.com/v1/chat/completions', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + apiKey },
          body: JSON.stringify({
            model: 'gpt-4o-mini',
            messages: [
              { role: 'system', content: 'You are a professional resume writer. Always respond with valid JSON only, no markdown.' },
              { role: 'user', content: prompt }
            ],
            temperature: 0.7, max_tokens: 2000
          })
        });
        if (!openaiRes.ok) {
          var openaiErr = await openaiRes.json();
          throw new Error((openaiErr.error && openaiErr.error.message) || 'OpenAI API error');
        }
        var openaiData = await openaiRes.json();
        content = openaiData.choices[0].message.content.trim();
      }

      content = content.replace(/^```json\n?/, '').replace(/^```\n?/, '').replace(/\n?```$/, '').trim();
      var resumeData = JSON.parse(content);
      renderResume(data.personal, resumeData);
      showToast('Resume generated successfully! 🎉', 'success');
      gaEvent('resume_generated', { method: currentProvider, tone: data.tone }); // ✅ GA Event

    } catch(err) {
      console.error(err);
      showToast('Error: ' + err.message, 'error');
    } finally {
      showLoading(false);
    }
  }

  function buildLocalResume(data) {
    var name = data.personal.firstName + ' ' + data.personal.lastName;
    var title = data.personal.jobTitle || 'Professional';
    var tone = data.tone || 'professional';

    var summaryPhrases = {
      professional: 'results-driven and highly motivated',
      creative: 'innovative and creative',
      technical: 'technically skilled and detail-oriented',
      concise: 'accomplished and impactful'
    };
    var adj = summaryPhrases[tone] || 'experienced';

    var skillStr = data.skills.length > 0 ? ' with expertise in ' + data.skills.slice(0,3).join(', ') : '';
    var targetStr = data.targetRole ? ' seeking to excel as ' + data.targetRole : '';
    var summary = name + ' is a ' + adj + ' ' + title + skillStr + targetStr + '. ' +
      'Demonstrates a strong track record of delivering high-quality results in fast-paced environments. ' +
      'Committed to continuous professional development and bringing measurable value to every role. ' +
      'Adept at collaborating with cross-functional teams to achieve organizational goals.';

    var bulletTemplates = [
      ['Led and managed key projects, delivering results on time and within budget',
       'Collaborated with cross-functional teams to drive process improvements',
       'Analyzed data and metrics to inform strategic decision-making',
       'Mentored junior team members and fostered a culture of excellence'],
      ['Developed and implemented innovative solutions that improved efficiency by 20%+',
       'Spearheaded initiatives that directly contributed to revenue growth',
       'Built and maintained strong relationships with stakeholders',
       'Streamlined workflows to reduce operational overhead']
    ];

    var experiences = data.experiences.map(function(e, i) {
      var customBullets = [];
      if (e.desc && e.desc.trim()) {
        var sentences = e.desc.split(/[.\n]+/).filter(function(s) { return s.trim().length > 10; });
        customBullets = sentences.slice(0, 3).map(function(s) {
          var s2 = s.trim();
          return s2.charAt(0).toUpperCase() + s2.slice(1) + (s2.endsWith('.') ? '' : '.');
        });
      }
      var template = bulletTemplates[i % 2];
      var bullets = customBullets.length >= 2 ? customBullets : template.slice(0, 3);
      return {
        title: e.title || 'Professional',
        company: e.company || 'Company',
        period: (e.start || '') + (e.end ? ' – ' + e.end : ''),
        location: e.location || '',
        bullets: bullets
      };
    });

    var education = data.educations.map(function(e) {
      return {
        degree: e.degree || 'Degree',
        school: e.school || 'Institution',
        year: e.year || '',
        note: e.achievements || ''
      };
    });

    return {
      summary: summary,
      experiences: experiences,
      education: education,
      skills: data.skills,
      languages: data.languages || '',
      certifications: data.certifications || ''
    };
  }

  function renderResume(personal, ai) {
    var output = document.getElementById('resume-output');
    var contactItems = [
      personal.email ? '<span><i class="fas fa-envelope"></i> ' + personal.email + '</span>' : '',
      personal.phone ? '<span><i class="fas fa-phone"></i> ' + personal.phone + '</span>' : '',
      personal.location ? '<span><i class="fas fa-map-marker-alt"></i> ' + personal.location + '</span>' : '',
      personal.linkedin ? '<span><i class="fab fa-linkedin"></i> ' + personal.linkedin + '</span>' : '',
      personal.website ? '<span><i class="fas fa-globe"></i> ' + personal.website + '</span>' : ''
    ].filter(Boolean).join('');

    var expHTML = (ai.experiences || []).map(function(exp) {
      return '<div class="resume-item">' +
        '<div class="resume-item-header">' +
          '<div class="resume-item-title">' + exp.title + '</div>' +
          '<div class="resume-item-date">' + (exp.period || '') + '</div>' +
        '</div>' +
        '<div class="resume-item-subtitle">' + exp.company + (exp.location ? ' &middot; ' + exp.location : '') + '</div>' +
        '<div class="resume-item-desc"><ul>' + (exp.bullets || []).map(function(b) { return '<li>' + b + '</li>'; }).join('') + '</ul></div>' +
      '</div>';
    }).join('');

    var eduHTML = (ai.education || []).map(function(edu) {
      return '<div class="resume-item">' +
        '<div class="resume-item-header">' +
          '<div class="resume-item-title">' + edu.degree + '</div>' +
          '<div class="resume-item-date">' + (edu.year || '') + '</div>' +
        '</div>' +
        '<div class="resume-item-subtitle">' + edu.school + '</div>' +
        (edu.note ? '<div class="resume-item-desc">' + edu.note + '</div>' : '') +
      '</div>';
    }).join('');

    var skillsHTML = (ai.skills || []).map(function(s) {
      return '<span class="resume-skill-pill">' + s + '</span>';
    }).join('');

    output.innerHTML =
      '<div class="resume-header">' +
        '<div class="resume-name">' + personal.firstName + ' ' + personal.lastName + '</div>' +
        (personal.jobTitle ? '<div class="resume-title">' + personal.jobTitle + '</div>' : '') +
        '<div class="resume-contact">' + contactItems + '</div>' +
      '</div>' +
      (ai.summary ? '<div class="resume-section"><div class="resume-section-title">Professional Summary</div><div class="resume-summary">' + ai.summary + '</div></div>' : '') +
      (expHTML ? '<div class="resume-section"><div class="resume-section-title">Work Experience</div>' + expHTML + '</div>' : '') +
      (eduHTML ? '<div class="resume-section"><div class="resume-section-title">Education</div>' + eduHTML + '</div>' : '') +
      (skillsHTML ? '<div class="resume-section"><div class="resume-section-title">Skills</div><div class="resume-skills-grid">' + skillsHTML + '</div></div>' : '') +
      (ai.languages ? '<div class="resume-section"><div class="resume-section-title">Languages</div><div class="resume-summary">' + ai.languages + '</div></div>' : '') +
      (ai.certifications ? '<div class="resume-section"><div class="resume-section-title">Certifications</div><div class="resume-summary">' + ai.certifications + '</div></div>' : '');

    document.getElementById('preview-placeholder').style.display = 'none';
    output.style.display = 'block';
    document.getElementById('action-bar').classList.add('visible');
  }

  function showLoading(show) {
    document.getElementById('loading-overlay').classList.toggle('active', show);
  }

  function toggleApiVisibility() {
    var input = document.getElementById('api-key');
    var icon = document.getElementById('eye-icon');
    if (input.type === 'password') { input.type = 'text'; icon.className = 'fas fa-eye-slash'; }
    else { input.type = 'password'; icon.className = 'fas fa-eye'; }
  }

  function copyResumeText() {
    var text = document.getElementById('resume-output').innerText;
    navigator.clipboard.writeText(text).then(function() {
      showToast('Resume text copied to clipboard!', 'success');
      gaEvent('copy_resume'); // ✅ GA Event
    });
  }

  function showToast(msg, type) {
    type = type || 'success';
    var toast = document.getElementById('toast');
    var toastMsg = document.getElementById('toast-msg');
    var icon = toast.querySelector('i');
    toastMsg.textContent = msg;
    toast.className = 'toast ' + type;
    icon.className = type === 'success' ? 'fas fa-check-circle' : 'fas fa-exclamation-circle';
    toast.classList.add('show');
    setTimeout(function() { toast.classList.remove('show'); }, 3500);
  }
